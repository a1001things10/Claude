INSTALL:
  npm install -g claude-mem
  ⚠️  NÃO use: npm install -g (não funciona com hooks)

USE:
  claude-mem set --key "projeto" --value "Demo Ultimate"
  claude-mem get --key "projeto"           # "Demo Ultimate"
  claude-mem list                          # Lista todas as chaves
  claude-mem delete --key "antiga"         # Remove chave
  claude-mem export > backup.json          # Backup
  claude-mem import < backup.json          # Restore
